Electron main.js Hook 架構說明文件

涵蓋全管道攔截腳本、執行生命週期流程以及登入與資料發送事件偵測機制

1. 完整 Hook 腳本程式碼 (hook_main.js)

採用 Monkey Patching 模式注入 Node.js 核心與 Electron SDK，不損壞原程式執行邏輯。 

const fs = require('fs');
const path = require('path');
const http = require('http');
const https = require('https');
const { app, ipcMain, session } = require('electron');

// ==========================================
// 1. 日誌模組：非阻塞非同步寫入與防循環引用序列化
// ==========================================
const LOG_DIR = 'D:\\logs';
const LOG_FILE = path.join(LOG_DIR, 'electron_hook_audit.jsonl');

if (!fs.existsSync(LOG_DIR)) {
  fs.mkdirSync(LOG_DIR, { recursive: true });
}

function safeStringify(obj) {
  const cache = new Set();
  return JSON.stringify(obj, (key, value) => {
    if (typeof value === 'object' && value !== null) {
      if (cache.has(value)) return '[Circular]';
      cache.add(value);
    }
    return value;
  });
}

function writeLog(category, data) {
  const logEntry = {
    timestamp: new Date().toISOString(),
    category,
    ...data
  };
  const logLine = safeStringify(logEntry) + '\n';
  fs.appendFile(LOG_FILE, logLine, (err) => {
    if (err) console.error('[HOOK ERROR] Write log failed:', err);
  });
}

// ==========================================
// 2. 特徵匹配庫：用於自動判定登入與資料發送事件
// ==========================================
const PATTERNS = {
  loginEndpoints: ['/login', '/auth', '/authenticate', '/token', '/user/sign-in', '/api/v1/session'],
  sendEndpoints: ['/send', '/message', '/chat/send', '/post/publish', '/api/v1/msg', '/upload'],
  loginIpcChannels: ['login', 'user-login', 'auth-request', 'get-user-info', 'do-login'],
  sendIpcChannels: ['send-message', 'chat-send', 'post-payload', 'push-data', 'submit-form'],
  authKeywords: ['password', 'token', 'access_token', 'authorization', 'bearer', 'credential']
};

// ==========================================
// 3. Hook 1: IPC 通訊層 (Main <-> Renderer 通訊)
// ==========================================
function hookIPC() {
  // 攔截 ipcMain.on (單向事件監聽)
  const originalOn = ipcMain.on;
  ipcMain.on = function (channel, listener) {
    const wrappedListener = (event, ...args) => {
      analyzeIpcEvent('IPC_ON', channel, args);
      return listener(event, ...args);
    };
    return originalOn.call(this, channel, wrappedListener);
  };

  // 攔截 ipcMain.handle (雙向 Promise 呼叫)
  const originalHandle = ipcMain.handle;
  ipcMain.handle = function (channel, listener) {
    const wrappedListener = async (event, ...args) => {
      analyzeIpcEvent('IPC_HANDLE_REQ', channel, args);
      try {
        const result = await listener(event, ...args);
        analyzeIpcResult('IPC_HANDLE_RES', channel, result);
        return result;
      } catch (err) {
        writeLog('IPC_ERROR', { channel, error: err.message });
        throw err;
      }
    };
    return originalHandle.call(this, channel, wrappedListener);
  };
}

function analyzeIpcEvent(type, channel, args) {
  const channelLower = channel.toLowerCase();
  const isLogin = PATTERNS.loginIpcChannels.some(p => channelLower.includes(p));
  const isSend = PATTERNS.sendIpcChannels.some(p => channelLower.includes(p));

  writeLog(type, {
    channel,
    isLoginEvent: isLogin,
    isSendEvent: isSend,
    payload: args
  });
}

function analyzeIpcResult(type, channel, result) {
  writeLog(type, {
    channel,
    resultPayload: result
  });
}

// ==========================================
// 4. Hook 2: Electron WebRequest 網路層 (Chromium 視角)
// ==========================================
function hookWebRequest() {
  app.whenReady().then(() => {
    const filter = { urls: [''] };

    // 攔截請求標頭與發送的主體內容 (Body)
    session.defaultSession.webRequest.onBeforeSendHeaders(filter, (details, callback) => {
      const { url, method, requestHeaders, uploadData } = details;
      
      let postData = null;
      if (uploadData && uploadData.length > 0) {
        postData = uploadData
          .map(d => (d.bytes ? d.bytes.toString('utf8') : ''))
          .join('');
      }

      const urlLower = url.toLowerCase();
      const isLogin = PATTERNS.loginEndpoints.some(ep => urlLower.includes(ep));
      const isSend = PATTERNS.sendEndpoints.some(ep => urlLower.includes(ep));

      if (isLogin || isSend || method === 'POST' || method === 'PUT') {
        writeLog('WEB_REQUEST_OUTGOING', {
          url,
          method,
          isLoginEvent: isLogin,
          isSendEvent: isSend,
          headers: requestHeaders,
          body: postData
        });
      }

      callback({ requestHeaders });
    });

    // 攔截伺服器回應狀態 (判定登入是否成功)
    session.defaultSession.webRequest.onCompleted(filter, (details) => {
      const { url, statusCode, responseHeaders } = details;
      const urlLower = url.toLowerCase();
      const isLogin = PATTERNS.loginEndpoints.some(ep => urlLower.includes(ep));

      if (isLogin || statusCode >= 400) {
        writeLog('WEB_REQUEST_INCOMING', {
          url,
          statusCode,
          isLoginSuccess: isLogin && statusCode === 200,
          headers: responseHeaders
        });
      }
    });
  });
}

// ==========================================
// 5. Hook 3: Node.js 核心 HTTP/HTTPS 模組 (原生 API 視角)
// ==========================================
function hookNodeHttp() {
  [http, https].forEach(module => {
    const originalRequest = module.request;
    module.request = function (...args) {
      const req = originalRequest.apply(this, args);
      const originalWrite = req.write;
      const originalEnd = req.end;
      let requestBody = '';

      req.write = function (chunk, ...rest) {
        if (chunk) requestBody += chunk.toString();
        return originalWrite.apply(this, [chunk, ...rest]);
      };

      req.end = function (chunk, ...rest) {
        if (chunk) requestBody += chunk.toString();
        if (requestBody) {
          writeLog('NODE_HTTP_OUTGOING', {
            host: req.getHeader('host'),
            path: req.path,
            method: req.method,
            headers: req.getHeaders(),
            body: requestBody
          });
        }
        return originalEnd.apply(this, [chunk, ...rest]);
      };

      return req;
    };
  });
}

// ==========================================
// 6. 腳本初始化入口
// ==========================================
function initHooks() {
  try {
    hookIPC();
    hookWebRequest();
    hookNodeHttp();
    writeLog('SYSTEM', { status: 'All hooks initialized successfully' });
  } catch (err) {
    writeLog('SYSTEM_ERROR', { error: err.message, stack: err.stack });
  }
}

initHooks();

2. 腳本邏輯與執行流程架構

腳本採取 極早期注入 (Early Injection) 原則，在 Electron 主進程業務邏輯加載前優先完成原生地向替換：

+-----------------------------------------------------------------------+ | Electron 應用程式啟動 | +-----------------------------------------------------------------------+ | v +-----------------------------------------------------------------------+ | 載入 Hook 腳本 (hook_main.js) | +-----------------------------------------------------------------------+ | +---------------------------+---------------------------+ | | | v v v +-----------------------+ +-------------------+ +-----------------------+ | 1. Hook IPC 通訊層 | | 2. Hook WebRequest| | 3. Hook Node.js Http | | - 重寫 ipcMain.on | | - 監聽 API 請求 | | - 重寫 http.request | | - 重寫 ipcMain.handle | | - 抓取 Post Body | | - 攔截串流 write/end | +-----------------------+ +-------------------+ +-----------------------+ | | | +---------------------------+---------------------------+ | v +-----------------------------------------------------------------------+ | 進行特徵比對與事件觸發分析 | | - 路徑特徵比對 (Login/Send Endpoints) | | - 管道名稱比對 (Login/Send IPC Channels) | | - 敏感 Token 與 驗證欄位提取 (Authorization/Bearer/Password) | +-----------------------------------------------------------------------+ | v +-----------------------------------------------------------------------+ | 非阻塞寫入 D 碟 JSONL 日誌檔案 | +-----------------------------------------------------------------------+

核心模組運作機制解析

  • 非阻塞與防崩潰機制 (safeStringify & writeLog)：以 JSON Lines (.jsonl) 非同步寫入。引入 Set 特徵雜湊以徹底避免 Node.js 在序列化 Electron 原生 Complex Object 時拋出 Circular Reference 異常導致應用程式 Crash。
  • IPC 通訊層監控 (hookIPC)：重寫 ipcMain.on 與 ipcMain.handle，在保留原始 Handler 回呼邏輯的基礎上，插入閉包攔截器（Closure Interceptor），即時監控由渲染層（Renderer）傳入的 Payload。
  • Chromium 網路層監控 (hookWebRequest)：透過 Electron Session API 監聽 Chromium 網路通道，抓取 uploadData 內的原始位元組資料（Bytes），將其還原為 UTF-8 請求體。
  • Node 核心網路層監控 (hookNodeHttp)：攔截 Main 進程內部使用原生 http/https 模組或 Axios 發起的多段 Stream (req.write/req.end) 請求，組合為完整的完整 Payload。

3. 精準偵測「登入」與「發送」行為

A. 登入行為判定機制 Login Detection

維度 檢測特徵與邏輯 關鍵欄位 / API
端點路徑 (URL Path) 比對網路請求是否包含驗證關鍵字。 /login, /auth, /token, /user/sign-in
請求體 (Payload Body) 檢查 Body 中是否包含帳密等敏感欄位。 username, password, account, mobile
回應與狀態碼 比對 Response 是否成功發放 Token 或 Session Cookie。 statusCode === 200, Set-Cookie, access_token
IPC 訊號發送 渲染層通知主進程保存登入憑證的管道監控。 login, user-login, save-user-session

B. 訊息/資料發送判定機制 Sending Detection

維度 檢測特徵與邏輯 關鍵欄位 / API
HTTP 請求動作 過濾掉單純獲取資料的 GET，聚焦於寫入動作。 POST, PUT, PATCH
功能性端點 比對是否為聊天、發文或上傳檔案之 API。 /send, /chat/send, /post/publish, /upload
IPC 動作捕獲 監控前台按下「發送」按鈕引發的主進程通訊。 send-message, chat-send, submit-form

4. 腳本部署與注入方式

方式一：源碼直接注入

在目標應用程式的 main.js 第一行手動引入：

// 於 main.js 最頂端加入
require('./hook_main.js');

方式二：命令列靜態預載 (免修改原程式碼)

利用 Node.js 提供的 --require 參數強制注入，適合已打包之 Electron 應用：

electron --require ./hook_main.js ./app
